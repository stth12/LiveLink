cmake_minimum_required(VERSION 3.22)
project(LiveLink VERSION 0.1.0 LANGUAGES C CXX)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "Windows targets require Windows. Build ios/ on macOS.")
endif()
set(JUCE_DIR "" CACHE PATH "JUCE source checkout")
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
else()
  add_subdirectory("${JUCE_DIR}" JUCE)
endif()
add_library(llcore INTERFACE)
target_include_directories(llcore INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(llcore INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(llcore INTERFACE ws2_32 advapi32)
juce_add_console_app(LiveLinkService PRODUCT_NAME "Live Link Service" ICON_BIG "installer/assets/LiveLink.png")
set_target_properties(LiveLinkService PROPERTIES OUTPUT_NAME "Live Link Service")
target_sources(LiveLinkService PRIVATE service/src/Main.cpp)
target_link_libraries(LiveLinkService PRIVATE llcore juce::juce_core dnsapi)
juce_add_gui_app(LiveLinkDesktop PRODUCT_NAME "Live Link" ICON_BIG "installer/assets/LiveLink.png")
set_target_properties(LiveLinkDesktop PROPERTIES OUTPUT_NAME "Live Link")
target_sources(LiveLinkDesktop PRIVATE desktop/src/Main.cpp)
target_link_libraries(LiveLinkDesktop PRIVATE llcore juce::juce_gui_basics)
juce_add_plugin(LiveLinkPlugin COMPANY_NAME "Live Link" BUNDLE_ID "audio.livelink.plugin" PLUGIN_MANUFACTURER_CODE LvLk PLUGIN_CODE Lv01 FORMATS VST3 PRODUCT_NAME "Live Link" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(LiveLinkPlugin PRIVATE plugin/src/Plugin.cpp)
target_link_libraries(LiveLinkPlugin PRIVATE llcore juce::juce_audio_utils)
enable_testing()
add_executable(LiveLinkTests tests/unit/CoreTests.cpp)
target_link_libraries(LiveLinkTests PRIVATE llcore)
add_test(NAME core COMMAND LiveLinkTests)
add_executable(LiveLinkProbe tests/integration/Probe.cpp)
target_link_libraries(LiveLinkProbe PRIVATE llcore juce::juce_core)
add_executable(LiveLinkReceiverTests tests/unit/ReceiverTests.cpp)
target_link_libraries(LiveLinkReceiverTests PRIVATE llcore)
add_test(NAME receiver COMMAND LiveLinkReceiverTests)
juce_add_console_app(LiveLinkPluginTests PRODUCT_NAME "LiveLinkPluginTests")
target_sources(LiveLinkPluginTests PRIVATE tests/unit/PluginTests.cpp plugin/src/Plugin.cpp)
target_link_libraries(LiveLinkPluginTests PRIVATE llcore juce::juce_audio_utils)
set_target_properties(LiveLinkPluginTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_test(NAME processor COMMAND LiveLinkPluginTests)
juce_add_console_app(LiveLinkVSTHostTest PRODUCT_NAME "LiveLinkVSTHostTest")
target_sources(LiveLinkVSTHostTest PRIVATE tests/integration/VSTHostTest.cpp)
target_compile_definitions(LiveLinkVSTHostTest PRIVATE JUCE_PLUGINHOST_VST3=1)
target_link_libraries(LiveLinkVSTHostTest PRIVATE llcore juce::juce_audio_utils)
set_target_properties(LiveLinkVSTHostTest PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(LiveLinkNetworkStress tests/performance/NetworkStress.cpp)
target_link_libraries(LiveLinkNetworkStress PRIVATE llcore juce::juce_core)
